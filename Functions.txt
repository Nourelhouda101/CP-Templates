ll gcdd (ll a, ll b) {                  //* GCD
    if (a == 0) return b;
    if (b > a) swap(a, b);
    return gcdd(a % b, b);
}

ll lcmm (ll a, ll b) {                 //* LCM
    return (a * b) / gcdd(a, b); 
}

vector <bool> sieves(1000001, 1);       //* Sieve (Primes From 1 to n) O(n*log(log(n)))
void sieve (ll n) {
    sieves[0] = sieves[1] = 0;

    for (ll i = 2; i * i <= n; i++) {
        if (sieves[i]) {
            for (ll j = i * i; j <= n; j += i)
            sieves[j] = 0;
        }
    }
}

ll phi(ll n) {     //* Phi, calculate coprimes less than n
    ll res = n;
    for (ll i = 2; i * i <= n; i++) {

        if (n % i == 0) {
           while (n % i == 0)
           n /= i;
   
           res -= res / i;
        }
    }

    if (n > 1)
    res -= res / n;

    return res;
}

ll Euler_totient (ll n) {
    float res = n;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            while(n % i == 0)
            n /= i;

            res *= (1.0 - (1.0 / (float)i));
        }
    }
    if (n > 1)
    res *= (1.0 - (1.0 / (float)n));
    return (ll)res;
}

const int LOG = 20;                                    //*SPARSE TABLE
int sparse_min[N][LOG], sparse_max[N][LOG], sparse_gcd[N][LOG], a[N];
int log_table[N];
void build_sparse(int n) {
    // log table
    log_table[1] = 0;
    for (int i = 2; i <= n; i++)
        log_table[i] = log_table[i / 2] + 1;

    // initialize base case
    for (int i = 0; i < n; i++) {
        sparse_min[i][0] = a[i];
        sparse_max[i][0] = a[i];
        sparse_gcd[i][0] = a[i];
    }

    // build
    for (int j = 1; (1 << j) <= n; j++) {
        for (int i = 0; i + (1 << j) <= n; i++) {
            sparse_min[i][j] = min(sparse_min[i][j - 1], sparse_min[i + (1 << (j - 1))][j - 1]);
            sparse_max[i][j] = max(sparse_max[i][j - 1], sparse_max[i + (1 << (j - 1))][j - 1]);
            sparse_gcd[i][j] = __gcd(sparse_gcd[i][j - 1], sparse_gcd[i + (1 << (j - 1))][j - 1]);

        }
    }
}

int query_gcd(int l, int r) {
    int len = r - l + 1;
    int j = log_table[len];
    return __gcd(sparse_gcd[l][j], sparse_gcd[r - (1 << j) + 1][j]);
}

int get_min(int l, int r) {
    int j = log_table[r - l + 1];
    return min(sparse_min[l][j], sparse_min[r - (1 << j) + 1][j]);
}

int get_max(int l, int r) {
    int j = log_table[r - l + 1];
    return max(sparse_max[l][j], sparse_max[r - (1 << j) + 1][j]);
}

ll extended_Euclidean_rec (ll a, ll b, ll& x, ll& y) {   
    if (b == 0) {  // you can use it to get the inverse if m isn't prime so
        x = 1;     //  you can't apply fermat's little theorem, consider a is the
        y = 0;     //  inverse, b is m and (x + m % m) % m is (a^-1)
        return a;
    }

    ll x1, y1;
    ll d = extended_Euclidean_rec(b, a % b, x1, y1);
    x = y1;
    y = x1 - y1 * (a / b);
    return d;
}

ll mod_inverse(ll b, ll m) {
    ll x, y;
    ll g = extended_Euclidean_rec(b, m, x, y);
    if (g != 1)
        return -1;  // inverse doesn't exist
    return (x % m + m) % m;  // ensure positive result
}


ll mod_divide(ll a, ll b, ll m) {
    a %= m;
    ll inv = mod_inverse(b, m);
    if (inv == -1) {
        cout << "Division not defined (no inverse)\n";
        return -1;
    }
    return (a * inv) % m;
}

ll mod_pow(ll base, ll exp, ll mod) {
    ll res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1)
            res = res * base % mod;
        base = base * base % mod;
        exp /= 2;
    }
    return res;
}

vector<ll> get_primes(ll limit) {  // limit is sqrt(r)
    vector <bool> is_prime(limit + 1, true);
    is_prime[0] = is_prime[1] = false;

    for (ll i = 2; i * i <= limit; i++) {
        if (is_prime[i]) {
            for (ll j = i * i; j <= limit; j += i)
                is_prime[j] = false;
        }
    }

    vector <ll> primes;
    for (ll i = 2; i <= limit; i++) {
        if (is_prime[i])
            primes.push_back(i);
    }
    return primes;
}

void segmented_sieve(ll l, ll r) {    //* segmented_sieve (l to r)
    if (l > r) return;

    ll limit = sqrt(r) + 1;
    vector<ll> primes = get_primes(limit);

    vector<bool> is_composite(r - l + 1, false);

    for (ll p : primes) {
        ll start = max(p * p, (l + p - 1) / p * p);
        for (ll j = start; j <= r; j += p) {
            is_composite[j - l] = true;
        }
    }

    if (l == 1)
        is_composite[0] = true;

    for (ll i = 0; i <= r - l; i++) {
        if (!is_composite[i])
            cout << (i + l) << " ";
    }
    cout << "\n";
}

ll count_factorial(ll n, ll p) {
    ll res = 0;
    while (n > 0) {
        res += n / p;
        n /= p;
    }
    return res;
}

ll cnt_dig(ll n) {
    ll power = 1, cnt = 0;
    for (int i = 1; i <= n; i++) {
        if (power * 10 > n)
        return cnt + (n - power + 1) * i;

        cnt += (power * 10 - power) * i;
        power *= 10;
    }
    return cnt;
}

// n is the number we wanna check, a ^ 2^s.d
bool check_composite(ll n, ll a, ll d, ll s) {
    ll x = modularexponentiation(a, d, n);

    if (x == 1 || x == n - 1) // then it's prime
    return 0;   // a^d = 1, a^2^s.d = -1 (+ n for % positive)

    for (ll r = 1; r < s; r++) {
        x = modmul(x, x, n); // a^2^s % d = n - 1 
        if (x == n - 1)
        return 0;
    }

    return 1;
}

bool miller_rabin(ll n, ll k) {
    if (n < 4)
    return (n == 2 || n == 3);

    ll s = 0, d = n - 1;

    while ((d & 1) == 0) { // d % 2 == 0
        d >>= 1;  // d /= 2
        s++;   // a^2^s.d = n - 1
    }

    for (int i = 0; i < k; i++) {
        ll a = 2 + rand() % (n - 2); // 2 <= a <= n - 2

        if (check_composite(n, a, d, s))
        return 0;
    }
    return 1;
}

bool hascycle = false;
void dfs(int node, vector<vector<int>>& adj, vector<int>& visited, stack<int>& st) {
    visited[node] = 1; // 1 = visiting

    for (int neighbor : adj[node]) {
        if (visited[neighbor] == 0)
            dfs(neighbor, adj, visited, st);
        else if (visited[neighbor] == 1)
            hascycle = true; // back edge → cycle
    }

    visited[node] = 2; // 2 = visited
    st.push(node);
}

void Dijkstra(int start, int t) {
    dis.assign(n + 1, LLONG_MAX);
    pre.assign(n + 1, -1);
    priority_queue<pair<ll, ll>, vector<pair<ll,ll>>, greater<pair<ll,ll>>>pq;
    //stores the cost of each node and the node itself in ascending order.
    pq.push({0, start});
    dis[start] = 0;
    pre[start] = start;

    while(!pq.empty()) {
        auto [a, b] = pq.top(); // cost, node
        pq.pop();

        if(dis[b] < a)  // if it's not the best path so far, don't walk through it.
        continue;

        for(auto [x, y] : adj[b]) {  // node and its cost.
            if(dis[b] + y < dis[x]) {  
            // if cost of the parent node + cost of the path to node x is less than the 
            //distance was calculatd for node x, then update x's cost with the better cost.
                dis[x] = dis[b] + y;
                pq.push({dis[x], x});
                pre[x] = b;
            }
        }
    }

    //find the path from start to t
    if(dis[t] == LLONG_MAX)
    return;

    ll cur = t;
    while(cur != start) {
        path.push_back(cur);
        cur = pre[cur]; // cur = the previous node of cur till we reach the start node.
    }

    path.push_back(start);
    reverse(all(path));
}

        if(rx - lx == 1) {
            if(lx < arr.size())
            segData[cur] = Node(arr[lx]);

            return; 
        }
 
        int mid = (rx + lx) / 2;
        init(arr, 2 * cur + 1, lx, mid);
        init(arr, 2 * cur + 2, mid, rx);
 
        segData[cur] = merge(segData[2 * cur + 1], segData[2 * cur + 2]);
    }
 
    void init(vector<ll> & arr) {
        init(arr, 0, 0, treeSize);
    }
 
    void set(int idx, ll val, int ni, int lx, int rx) {
        if(rx - lx == 1) { // leaf
            segData[ni].change(val);
            return;
        }
 
        int mid = (rx + lx) / 2;
 
        if(idx < mid)
        set(idx, val, 2 * ni + 1, lx, mid);
 
        else
        set(idx, val, 2 * ni + 2, mid, rx);
 
        segData[ni] = merge(segData[2 * ni + 1], segData[2 * ni + 2]);
    }
 
    void set(int idx, ll val) {
        set(idx, val, 0, 0, treeSize);
    }
 
    //l, r -> the range I'm asking about, lx, rx the range THAT node covers.
    Node get(int l, int r, int ni, int lx, int rx) {
        if(lx >= l && rx <= r) // the node is fully inside the range I'm asking about.
        return segData[ni];
 
        if(lx >= r || rx <= l) //the node is completely outside the range
        return Node(); //return neutral element (sum = 0).
 
        int mid = (lx + rx) / 2;
        Node left = get(l, r, 2 * ni + 1, lx, mid);
        Node right = get(l, r, ni * 2 + 2, mid, rx);
 
        return merge(left, right);
    }
 
    ll get(int l, int r) {
        return get(l, r, 0, 0, treeSize).sum;
    }

struct DSU {
    // par[u] < 0  ->  root, component size = -par[u]
    // par[u] >= 0 ->  non-root, parent = par[u]
    int par[N], ncmp;

    void init(int n) {
        memset(par, -1, n * sizeof par[0]); // -1 = root of size 1
        ncmp = n;
    }

    int find(int u) {
        return par[u] < 0 ? u : par[u] = find(par[u]); // path compression
    }

    // s = small, b = big (by convention - swap if inconsistent).
    // Returns false if s and b are already in the same component.
    bool join(int s, int b) {
        s = find(s); b = find(b);
        if (s == b) return false;
        if (par[b] > par[s]) swap(s, b); // par more negative = bigger; ensure b is bigger
        par[b] += par[s];                // merge sizes (both negative -> sum more negative)
        par[s] = b;                      // s becomes child of b
        ncmp--;
        return true;
    }
};
